31. def parse_functions(filename: str):
32.     """The function takes a Python filename and reads it to find the functions defined inside.
33.     It returns a tuple of tuples, each containing the function’s starting line number,
34.     name, parameter list as text, and full code as text with empty lines and comments removed.
35.     """
36.     try:
37.         with open(filename, 'r') as input_file:
38.             lines = input_file.readlines()
39.     except FileNotFoundError:
40.         print("The input file does not exist.")
41.         raise
42. 
43.     parsed_f = []
44.     i = 0
45.     total = len(lines)
46. 
47.     while i < total:
48.         line = lines[i]
49.         stripped = line.strip()
50. 
51.         if stripped.startswith("def ") and "(" in stripped and ":" in stripped:
52.             start = i + 1
53.             indent = len(line) - len(line.lstrip())
54. 
55.             def_pos = line.find("def ")
56.             paren_start = line.find("(", def_pos)
57.             paren_end = line.rfind(")")
58. 
59.             f_name = line[def_pos +4:paren_start].strip()
60.             params = line[paren_start +1:paren_end].strip()
61. 
62.             code_ls = []
63.             j = i 
64.             while j < total:
65.                 current_line = lines[j]
66.                 current_stripped = current_line.strip()
67. 
68.                 if j > i and current_stripped:
69.                     current_indent = len(current_line) - len(current_line.lstrip())
70.                     if current_indent <= indent:
71.                         break 
72. 
73.                 if current_stripped and not current_stripped.startswith("#"):
74.                     code_ls.append(current_line.rstrip())
75. 
76.                 j += 1
77. 
78.             final_code = "\n".join(code_ls)
79.             parsed_f.append((start, f_name, params, final_code))
80.             i = j - 1
81. 
82.         i += 1 
83.     parsed_f.sort(key=lambda item: item[1])
84.     return tuple(parsed_f)
